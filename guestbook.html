Simulated Database Entries

PHP + MySQL Code That Would Save This to the Database

<?php
$servername = "localhost";
$username = "your_db_username";
$password = "your_db_password";
$dbname = "guestbook_app";

$conn = new mysqli($servername, $username, $password, $dbname);

if ($_SERVER["REQUEST_METHOD"] == "POST") {
    $name = $_POST['name'];
    $message = $_POST['message'];

    $sql = "INSERT INTO guestbook_entries (name, message)
            VALUES ('$name', '$message')";

    if ($conn->query($sql) === TRUE) {
        echo "Input was sent to the database.";
    } else {
        echo "Error: " . $conn->error;
    }
}

$result = $conn->query("SELECT * FROM guestbook_entries ORDER BY submitted_at DESC");

while($row = $result->fetch_assoc()) {
    echo "<div class='entry'>";
    echo "<p><strong>" . htmlspecialchars($row['name']) . "</strong> wrote:</p>";
    echo "<p>" . nl2br(htmlspecialchars($row['message'])) . "</p>";
    echo "<small>Posted on " . $row['submitted_at'] . "</small>";
    echo "</div>";
}
?>
